Since your folder is:

~/all23_files

and your script is index.js, run these commands.

1. Go to the job folder
cd ~/all23_files
ls

You should see:

Dockerfile  index.js  package.json

2. Make sure Dockerfile uses index.js
cat > Dockerfile <<'EOF'
FROM node:20-bookworm

WORKDIR /app

COPY package*.json ./
RUN npm install

COPY . .

CMD ["node", "index.js"]
EOF


3. Install locally and test small first

rm -rf node_modules package-lock.json
npm install
run:

cd ~/all23_files

BUCKET_NAME=all23_06252026 \
LIMIT=1017 \
BATCH_SIZE=10 \
HEADER_READ_BYTES=131072 \
node index.js

4. run as Cloud Run and close Shell:

gcloud builds submit --tag gcr.io/personalgenomes/all23-06252026-job .

gcloud run jobs create all23-06252026-job \
  --image gcr.io/personalgenomes/all23-06252026-job \
  --region us-east4 \
  --memory 4Gi \
  --cpu 2 \
  --task-timeout 3600 \
  --command node \
  --args index.js \
  --set-env-vars BUCKET_NAME=all23_06252026,LIMIT=1017,BATCH_SIZE=10,HEADER_READ_BYTES=131072

Run without waiting:

gcloud run jobs execute all23-06252026-job \
  --region us-east4

Check later:

gcloud run jobs executions list \
  --job=all23-06252026-job \
  --region us-east4
